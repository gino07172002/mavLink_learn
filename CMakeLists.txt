cmake_minimum_required(VERSION 3.20)
project(mavlink_learning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mavlink/common/mavlink.h")
  message(FATAL_ERROR "Missing MAVLink headers. See third_party/README.md")
endif()
add_library(lab_support INTERFACE)
target_include_directories(lab_support INTERFACE common/include)
target_include_directories(lab_support SYSTEM INTERFACE third_party/mavlink)
if(WIN32)
  target_link_libraries(lab_support INTERFACE ws2_32)
endif()
if(MSVC)
  target_compile_options(lab_support INTERFACE /W4 /utf-8)
else()
  target_compile_options(lab_support INTERFACE -Wall -Wextra -Wpedantic)
endif()
# Keep executable paths stable for existing commands/scripts while sources live in each Lab.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(PX4LITE_ROOT "" CACHE PATH "Optional existing px4-lite_learning checkout")
enable_testing()
find_package(Python3 3.9 COMPONENTS Interpreter)
add_subdirectory(labs/lab04_bridge) # Exposes the toy Commander model used by protocol checks.
foreach(lab lab00_protocol lab01_telemetry lab02_command lab03_offboard lab05_sitl
    lab06_px4lite lab07_services lab08_router lab09_signing)
  add_subdirectory(labs/${lab})
endforeach()
if(Python3_FOUND)
  add_test(NAME reset_lab_utility COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reset_lab.py")
  add_test(NAME udp_integration COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE_DIR:lab04_bridge>")
  set_tests_properties(udp_integration PROPERTIES TIMEOUT 80)
endif()
